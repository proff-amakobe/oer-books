% Subtle original network artwork, anchored to a tcolorbox frame.
\newcommand{\EISBoxNetwork}{%
  \draw[EISCyan,opacity=.20,line width=.7pt]
    ([xshift=-2.8in,yshift=-.75in]frame.north east) coordinate (a)
    --++(-.75in,-.55in) coordinate (b)
    --++(-.85in,-.65in) coordinate (c);
  \draw[EISCyan,opacity=.20,line width=.7pt]
    (a)--++(-.2in,-1.35in) coordinate (d)--(c) (b)--++(.75in,-1.15in) coordinate (e)--(d);
  \foreach \p in {a,b,c,d,e}{\fill[EISCyan,opacity=.24] (\p) circle (3pt);}%
}
